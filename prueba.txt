//* PRUEBA GENERAL - Grupo 2 - Tema especial: PRINCIPIO *//

DECLARE.SECTION
[a1, b1, resul, nombre] := [FLOAT, INT, INT, STRING]
[p1, p2, p3] := [FLOAT, FLOAT, INT]
ENDDECLARE.SECTION

PROGRAM.SECTION

//* ---- Comentario con anidado de un nivel ----
IF (a1 <= 30) THEN
b1 ::= 1;  //* asignacion entera *//
ENDIF
*//

//* ---- Asignaciones y constantes ---- *//
a1 ::= 99999.99;
a1 ::= 99.;
a1 ::= .9999;
b1 ::= 32767;
b1 ::= 0;
nombre ::= "HOLA MUNDO";
nombre ::= "hola";
p1 ::= a1;
p3 ::= (b1 + 2) * 3 / 4 - 1;

//* ---- Salida ---- *//
WRITE "@sdADaSjfla%dfg";
WRITE 99.999;
WRITE 7;
WRITE a1;
write "ewr";       //* write en minuscula, como en el enunciado *//
write 99.999;
write b1;

//* ---- Decisiones y condiciones ---- *//
IF (a1 >= 5 AND b1 < 10) THEN
    WRITE "ok";
ELSE
    WRITE "no";
ENDIF

IF (a1 == 3 OR b1 != 4) THEN
    WRITE b1;
ENDIF

//* ---- Condicion con cada comparacion entre parentesis (formato del enunciado) ---- *//
IF ((a1 >= 5) AND (b1 < 10)) THEN
    WRITE "ok";
ENDIF

//* ---- Ciclo ---- *//
WHILE (b1 <= 10)
    b1 ::= b1 + 1;
ENDWHILE

//* ---- Tema especial: PRINCIPIO ---- *//
resul ::= PRINCIPIO(4;[10,20,30,40,5,4]);   //* La suma es: 100 *//
resul ::= PRINCIPIO(5;[2,2,2,4]);           //* La lista no tiene suficientes elementos *//
resul ::= PRINCIPIO(1;[2,1,1,4]);           //* La suma es: 2 *//
resul ::= PRINCIPIO(2;[10,20,30,40,50,40]); //* La suma es: 30 *//
resul ::= PRINCIPIO(1;[]);                  //* La lista esta vacia *//
resul ::= 4 + 3 + PRINCIPIO(4;[3,2,1,3,8]); //* PRINCIPIO dentro de una expresion *//
a1 ::= PRINCIPIO(2;[1.5,2,3.25]);         //* lista con reales: suma 3.5, se asigna a un FLOAT *//

ENDPROGRAM.SECTION
